\newglossaryentry{dispatchkey}{
  name=\texttt{DispatchKey},
  description={13 槽枚举（\texttt{p10/include/DispatchKey.h:15}），选择算子的哪一实现运行。后端(0--2)、\texttt{Autocast}(3--5)、\texttt{Autograd}(6--8)、\texttt{Vmap}(9--11)、\texttt{Composite}(12)。},
  sort=DispatchKey
}
\newglossaryentry{dispatcher}{
  name=\texttt{Dispatcher},
  description={将 \texttt{OperatorName} 映射到 \texttt{DispatchTable} 的静态注册表。内核槽数组为 \texttt{array<atomic<KernelFunction>,13>}，位于 \texttt{p10/include/Dispatcher.h:49--61}。},
  sort=Dispatcher
}
\newglossaryentry{tensoriterator}{
  name=\texttt{TensorIterator},
  description={七步广播/计算流水线（\texttt{coerce\_inputs}、\texttt{split\_inputs}、\texttt{analyze\_broadcast}、\texttt{compute\_strides}、\texttt{reorder\_dims}、\texttt{coalesce\_dims}、\texttt{allocate\_output}），位于 \texttt{p10/src/TensorIterator.cpp:1077}。},
  sort=TensorIterator
}
\newglossaryentry{sequence_nr}{
  name=\texttt{sequence\_nr},
  description={每线程单调计数器，在 \texttt{Node::Node}（\texttt{tpx/include/Node.h:50,109}）中赋值。由 \texttt{ReadyQueue}（\texttt{Engine.h:38}）用于确定性地优先重算。},
  sort=SequenceNR
}
\newglossaryentry{savedvariable}{
  name=\texttt{SavedVariable},
  description={类型擦除包装，保存张量及其 \texttt{version\_counter} 快照，解包时断言张量自前向以来未被原地修改。位于 \texttt{tpx/include/SavedVariable.h:15,33}。},
  sort=SavedVariable
}
\newglossaryentry{oprecord}{
  name=\texttt{OpRecord},
  description={栈分配的剖析记录，在每个分发算子上打戳，位于 \texttt{p10/include/Profiler.h:145}。持有 \texttt{begin/end\_ns}、\texttt{tid}、\texttt{kind}、可选 \texttt{shapes/dtypes}、\texttt{site\_id/stack\_id}。},
  sort=OpRecord
}
\newglossaryentry{gputimer}{
  name=\texttt{GpuTimerPair},
  description={池化的 \texttt{cudaEvent} 对，为每个分发的 CUDA 算子装备，位于 \texttt{p10/include/Profiler.h:227}。在会话结束时解析 \texttt{gpu\_ms}。宿主侧 \texttt{begin/end} 记于 \texttt{OpRecord}。},
  sort=GpuTimerPair
}
\newglossaryentry{redispatch}{
  name=重分发,
  description={算子在键 \emph{K} 上内部以更低优先级的 \emph{K'} 自调用的方案。被 \texttt{Autograd}（调用后端）、\texttt{Autocast}（调用默认类型）、\texttt{Vmap}（调用批量内核）使用。},
  sort=Redispatch
}
\newglossaryentry{composite}{
  name=\texttt{Composite},
  description={\texttt{DispatchKey} 的 12 号槽。后端无关回退；永不在 \texttt{DispatchKeySet} 中，仅由 \texttt{OperatorHandle::getKernel} 为 \texttt{is\_backend\_key} 查阅。},
  sort=Composite
}
\newglossaryentry{tp_packagedir}{
  name=\texttt{TP\_PACKAGE\_DIR},
  description={CMake 变量，将所有库的 \texttt{OUTPUT\_DIRECTORY} 固定到 \texttt{\$\{CMAKE\_CURRENT\_SOURCE\_DIR\}/tensorplay}。设于 \texttt{CMakeLists.txt:326,1170}。},
  sort=TPPackageDir
}
\newglossaryentry{usb_slot}{
  name=USB 槽,
  description={纯虚接口 \texttt{AutogradMetaBase}（\texttt{p10/include/AutogradMetaBase.h:13--25}），使 \texttt{p10} 无需包含 \texttt{tpx} 即可携带微分状态。具体插件为 \texttt{AutogradMeta}，位于 \texttt{tpx/include/AutogradMeta.h:26--108}。},
  sort=USBSlot
}
\newglossaryentry{linktime_axiom}{
  name=链接期无环性,
  description={公理：对任意 $(a,b)\in D$，\texttt{level}(a) > \texttt{level}(b)，其中 \texttt{level}(p10)=0、\texttt{level}(tpx)=1、\texttt{level}(stax)=2、\texttt{level}(\_C)=3，由 \texttt{CMakeLists.txt:938--944} 的 \texttt{add\_subdirectory} 顺序与仅向下的 \texttt{target\_link\_libraries} 边强制。},
  sort=LinkTimeAcyclicity
}
